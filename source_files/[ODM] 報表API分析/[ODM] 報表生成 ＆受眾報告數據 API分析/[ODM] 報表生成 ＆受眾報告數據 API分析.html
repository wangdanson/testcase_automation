      A. 點擊率計算 (CTR)
  def calculate_ctr(video_click, banner_click, tb, cv_imp)
    retval = ((video_click.to_f + banner_click.to_f + tb.to_f) / cv_imp.to_f).round(4)
    retval.nan? ? 0 : retval
  end

  # 應用場景:
  # ctr_cv: 基於 cv 的點擊率
  # ctr_imp: 基於 impression 的點擊率

  計算範例:
  # 資料: videoClick=45, bannerClick=12, tb=3, cv=1523
  ctr_cv = calculate_ctr(45, 12, 3, 1523)
  # = (45 + 12 + 3) / 1523 = 60 / 1523 = 0.0394 (3.94%)

  B. 互動率計算 (CPE Ratio)
  def calculate_cperatio(cpe, cv_imp)
    retval = (cpe.to_f / cv_imp.to_f).round(4)
    retval.nan? ? 0 : retval
  end

  # 應用場景:
  # cpeRatio_cv: 基於 cv 的互動率
  # cpeRatio_imp: 基於 impression 的互動率

  C. AER 比率計算
  def calculate_aerratio(video_click, banner_click, view5s, cpe, cv_imp)
    sum_stats = video_click.to_f + banner_click.to_f + view5s.to_f + cpe.to_f
    retval = (sum_stats / cv_imp.to_f).round(4)
    retval.nan? ? 0 : retval

      1. 智能依賴解析 - 系統自動分析用戶選擇的指標，找出計算所需的基礎數據

      2. 分層計算 - 先計算活動總計，再計算各表單分組，確保數據一致性

      3. 彈性公式支援 - ODM 和 SuperDSP 使用不同的計算公式，適應不同平台需求

      4. 優雅的排序機制 - 標準欄位按業務邏輯排序，互動明細按語言特性排序

      5. 高效過濾 - 只保留用戶選擇的欄位，移除計算用的中間數據

      處理流程

      1. 依賴分析 → 解析用戶選擇的指標，確定需要獲取和計算的欄位

      2. 基礎數據獲取 → 從 ClickHouse、第三方平台、互動明細獲取原始數據

      3. 衍生指標計算 → 使用預定義公式計算點擊率、互動率等衍生指標

      4. 資料結構整理 → 按用戶選擇過濾欄位，按業務邏輯排序

      5. 格式化輸出 → 轉換成 Excel 輸出所需的格式

        def sort_eng_col(undefined_col)
    # 定義權重, 越前面的順位權重越高
    sort_order = [
      { order: 1, condition: lambda { |key, _| key.match?(/\\A[\\p{Han}\\d]+\\z/) } }, # 中文字串
      { order: 2, condition: lambda { |key, _| key.match?('_') } },                # 有底線的字串  
      { order: 3, condition: lambda { |key, _| key.match?(/\\A[A-Za-z]+\\z/) } },    # 英文字串
    ]

    sorted_data = undefined_col.sort do |(key1, value1), (key2, value2)|
      order1 = sort_order.find { |rule| rule[:condition].call(key1, value1) }&.dig(:order) || Float::INFINITY
      order2 = sort_order.find { |rule| rule[:condition].call(key2, value2) }&.dig(:order) || Float::INFINITY
      order1 <=> order2
    end

    sorted_data.to_h
  end

          def rename_tab_names(data)
    exists_tab_names = { '總計' => 1 }  # 預設保留 '總計' 名稱

    data.each do |ind, v|
      tab_name = v[:tab_name]

      if exists_tab_names[tab_name]
        # 如果名稱重複，加上編號前綴
        v[:tab_name] = "(#{exists_tab_names[tab_name]}) #{tab_name} "
        v[:tab_name] = limit_tab_name(v[:tab_name]) if v[:tab_name].bytesize > 31
        exists_tab_names[tab_name] += 1
      else
        # 第一次出現的名稱，記錄到追蹤表中
        exists_tab_names[tab_name] = 1
      end
    end
  end

          def limit_tab_name(tab_name)
    final_tab_name = ""
    tab_name.each_char do |char|
      limit_bytesize = final_tab_name.bytesize + char.bytesize
      break if limit_bytesize > 31  # Excel 工作表名稱限制 31 bytes
      final_tab_name << char
    end
    final_tab_name
  end

    result = @lookup_index_exportname.map do |index, name|
      # 1. 移除 Excel 不允許的特殊字元
