  2. 查詢當天需建立sequnence的pre_campaign 

SELECT `pre_campaign_dailyreach`.* FROM `pre_campaign_dailyreach` 
INNER JOIN `pre_campaign_detail` ON `pre_campaign_detail`.`id` = `pre_campaign_dailyreach`.`pre_campaign_detail_id` 
INNER JOIN `pre_campaign` ON `pre_campaign`.`id` = `pre_campaign_detail`.`pre_campaign_id` 
WHERE `pre_campaign_dailyreach`.`c_date` = '2023-07-07' 
AND `pre_campaign_detail`.`enable` = 1 
AND (`pre_campaign_detail`.`pack_id` IS NOT NULL) 
AND `pre_campaign`.`status` = 'oncue' 
AND `pre_campaign`.`has_sequence` = 1
